Evidence behind the explanations in docs/benchmarks.md
=====================================================

This file holds what was recorded while looking for the reason of each row where shale is slower
than LevelDB: system call counts, the engines' own counters, and where the time of the calling
thread went. None of it is a result that the tables quote. The tables of docs/benchmarks.md are
made from the JSON files next to this one. What is here comes from separate runs of the same
workloads, on the same machine and the same Release build, made with the probe at the end of this
file (which calls shale_bench's own workload functions), while the machine was not held quiet. It
gives counts and shares. The speeds that the probe printed are not recorded, and are not quoted,
except in section 7.4, which compares runs of the probe with each other. The probe is compiled with
`c++ -O2 -g -std=c++20` and linked with the libraries of the Release build. Sections 0 to 6 are of the
build before the four changes; section 7 is of the builds after each of them, and says which.

How each part was made:

  * System calls: a library loaded with DYLD_INSERT_LIBRARIES that counts pread, read, write, mmap,
    munmap, fsync and fcntl(F_FULLFSYNC) (interpose.c, at the end of this file), and a snapshot of
    the counts around each benchmark (probe.cc, at the end of this file).
  * Counters: shale.stats and leveldb.stats, read through each engine's GetProperty after each
    benchmark. They count from the open of the database, and overwrite opens it again after its
    load, so overwrite's figures are the overwrites alone.
  * Where the time went: `sample <pid> <seconds>` on the probe, started a moment after the
    benchmark began, so that it samples the timed loop. Each thread is reported on its own. The
    percentages are shares of the samples of the thread that calls the engine (the one that runs
    the benchmark loop), by the function at the top of the stack. A share is of that thread's time
    in that run, and is not a time. "Compacted" means that the probe compacted the database between
    overwrite and the first read, as shale_bench --compact_before_reads does.

static void Snap(const std::string& label) {
  using Fn = void (*)(const char*);
  static Fn fn = reinterpret_cast<Fn>(dlsym(RTLD_DEFAULT, "interpose_snapshot"));
  if (fn) fn(label.c_str());
}

class ShaleProbe : public BenchEngine {
 public:
  std::string Name() const override { return "shale"; }
  Status Open(const std::string& path, bool fresh) override {
    Close();
    Options o;
    o.create_if_missing = true;
    if (fresh) {
      Status s = DestroyDB(path, o);
      if (!s.ok()) return s;
    }
    Status r = DB::Open(o, path, &db_);
    std::printf("MARK opened fresh=%d\n", fresh ? 1 : 0); std::fflush(stdout);
    return r;
  }
  Status Put(std::string_view k, std::string_view v, bool sync) override {
    WriteOptions w;
    w.sync = sync;
    return db_->Put(w, k, v);
  }
  Status Get(std::string_view k, std::string* v) override { return db_->Get(ReadOptions(), k, v); }
  Status Scan(std::string_view from, size_t limit, size_t* seen) override {
    std::unique_ptr<Iterator> it = db_->NewIterator(ReadOptions());
    if (from.empty()) it->SeekToFirst(); else it->Seek(from);
    size_t n = 0;
    for (; it->Valid() && (limit == 0 || n < limit); it->Next()) { (void)it->key(); (void)it->value(); n++; }
    *seen = n;
    return it->status();
  }
  Status CompactAll() override { return db_->CompactRange(std::nullopt, std::nullopt); }
  std::string WriteAmplification() override { std::string s; db_->GetProperty("shale.write-amplification", &s); return s; }
  void Close() override { db_.reset(); }
  std::string Props() {
    std::string s, t;
    db_->GetProperty("shale.stats", &s);
    return s;
  }
 private:
  std::unique_ptr<DB> db_;
};

static Status Conv(const leveldb::Status& s) {
  if (s.ok()) return Status::OK();
  if (s.IsNotFound()) return Status::NotFound("leveldb");
  return Status::IOError("leveldb", s.ToString());
}

class LevelDBProbe : public BenchEngine {
 public:
  LevelDBProbe() : filter_(leveldb::NewBloomFilterPolicy(10)) {
    o_.create_if_missing = true;
    o_.write_buffer_size = 4 << 20;
    o_.block_size = 4096;
    o_.filter_policy = filter_.get();
    o_.max_open_files = 1000;
    o_.compression = leveldb::kNoCompression;
  }
  std::string Name() const override { return "leveldb"; }
  Status Open(const std::string& path, bool fresh) override {
    Close();
    cache_.reset(leveldb::NewLRUCache(8 << 20));
    o_.block_cache = cache_.get();
    if (fresh) leveldb::DestroyDB(path, o_);
    leveldb::DB* db = nullptr;
    auto s = leveldb::DB::Open(o_, path, &db);
    db_.reset(db);
    std::printf("MARK opened fresh=%d\n", fresh ? 1 : 0); std::fflush(stdout);
    return Conv(s);
  }
  Status Put(std::string_view k, std::string_view v, bool sync) override {
    leveldb::WriteOptions w; w.sync = sync;
    return Conv(db_->Put(w, leveldb::Slice(k.data(), k.size()), leveldb::Slice(v.data(), v.size())));
  }
  Status Get(std::string_view k, std::string* v) override {
    leveldb::ReadOptions r; r.verify_checksums = true;
    return Conv(db_->Get(r, leveldb::Slice(k.data(), k.size()), v));
  }
  Status Scan(std::string_view from, size_t limit, size_t* seen) override {
    leveldb::ReadOptions r; r.verify_checksums = true;
    std::unique_ptr<leveldb::Iterator> it(db_->NewIterator(r));
    if (from.empty()) it->SeekToFirst(); else it->Seek(leveldb::Slice(from.data(), from.size()));
    size_t n = 0;
    for (; it->Valid() && (limit == 0 || n < limit); it->Next()) { (void)it->key(); (void)it->value(); n++; }
    *seen = n;
    return Conv(it->status());
  }
  Status CompactAll() override { db_->CompactRange(nullptr, nullptr); return Status::OK(); }
  std::string WriteAmplification() override { return "n/a"; }
  void Close() override { db_.reset(); }
  std::string Props() {
    std::string s;
    db_->GetProperty("leveldb.stats", &s);
    return s;
  }
 private:
  std::unique_ptr<const leveldb::FilterPolicy> filter_;
  leveldb::Options o_;
  std::unique_ptr<leveldb::Cache> cache_;
  std::unique_ptr<leveldb::DB> db_;
};

int main(int argc, char** argv) {
  if (argc < 5) { std::fprintf(stderr, "probe <shale|leveldb> <num> <db> <bench,bench...> [pause_ms_before_each]\n"); return 2; }
  const std::string engine_name = argv[1];
  WorkloadOptions options;
  options.num = std::strtoull(argv[2], nullptr, 10);
  options.sync_num = 10000;
  options.value_size = 100;
  options.seed = 301;
  options.db = argv[3];
  std::vector<Benchmark> list;
  { std::string l = argv[4]; size_t st = 0; while (true) { size_t c = l.find(',', st); auto n = FindBenchmark(l.substr(st, c == std::string::npos ? c : c - st)); if (!n) { std::fprintf(stderr, "bad benchmark\n"); return 2; } list.push_back(*n); if (c == std::string::npos) break; st = c + 1; } }
  const int pause_ms = argc > 5 ? std::atoi(argv[5]) : 0;
  const int loops = argc > 6 ? std::atoi(argv[6]) : 1;
  ShaleProbe sp; LevelDBProbe lp;
  BenchEngine* e = engine_name == "shale" ? static_cast<BenchEngine*>(&sp) : &lp;
  std::optional<Populated> source;
  bool compacted = false;
  for (Benchmark b : list) {
    RunResult r;
    Status s;
    if (pause_ms) std::this_thread::sleep_for(std::chrono::milliseconds(pause_ms));
    Snap(std::string("before ") + BenchmarkName(b));
    if (BenchmarkWrites(b)) {
      Populated p;
      s = RunWriteBenchmark(e, options, b, &p, &r);
      source = std::move(p);
    } else {
      if (std::getenv("PROBE_COMPACT") != nullptr && !compacted) {
        Status c = e->CompactAll();
        compacted = true;
        std::printf("MARK compacted %s\n", c.ok() ? "ok" : c.ToString().c_str()); std::fflush(stdout);
      }
      std::printf("MARK loop %s x%d\n", BenchmarkName(b), loops); std::fflush(stdout);
      Snap(std::string("loop start ") + BenchmarkName(b));
      for (int i = 0; i < loops; i++) {
        s = RunReadBenchmark(e, options, b, *source, &r);
        if (!s.ok()) break;
      }
      Snap(std::string("after ") + std::to_string(loops) + " x " + BenchmarkName(b));
    }
    if (!s.ok()) { std::printf("%s failed: %s\n", BenchmarkName(b), s.ToString().c_str()); return 1; }
    std::printf("== %s %s: %.0f ops/s, %.3f s, engine %.3f s, wa %s\n", engine_name.c_str(), BenchmarkName(b), r.OpsPerSecond(), r.seconds, r.engine_seconds, r.write_amplification ? r.write_amplification->c_str() : "-");
    if (r.latency) std::printf("   p50 %.2f p99 %.2f p99.9 %.2f us, avg %.2f us\n", r.latency->p50, r.latency->p99, r.latency->p999, r.latency->average);
    std::string props = engine_name == "shale" ? sp.Props() : lp.Props();
    std::printf("%s\n", props.c_str());
    if (engine_name == "shale") { std::string c; // counters
      /* shale.counters is private to stats text above */ }
  }
  return 0;
}

#include <fcntl.h>

static atomic_long fullfsync, fsync_calls, fcntl_other;
static atomic_long fullfsync_fail;
static atomic_long n_pread, n_read, n_write, n_pwrite, n_mmap, n_munmap;
static atomic_long b_pread, b_write;

#define DYLD_INTERPOSE(_replacement, _replacee) \
  __attribute__((used)) static struct { const void* replacement; const void* replacee; } \
  _interpose_##_replacee __attribute__((section("__DATA,__interpose"))) = { \
    (const void*)(unsigned long)&_replacement, (const void*)(unsigned long)&_replacee };

static int my_fcntl(int fd, int cmd, ...) {
  va_list ap;
  va_start(ap, cmd);
  long arg = va_arg(ap, long);
  va_end(ap);
  int r = fcntl(fd, cmd, arg);
  if (cmd == F_FULLFSYNC) {
    atomic_fetch_add(&fullfsync, 1);
    if (r != 0) atomic_fetch_add(&fullfsync_fail, 1);
  } else {
    atomic_fetch_add(&fcntl_other, 1);
  }
  return r;
}
static int my_fsync(int fd) {
  atomic_fetch_add(&fsync_calls, 1);
  return fsync(fd);
}
static ssize_t my_pread(int fd, void* b, size_t n, off_t o) { ssize_t r = pread(fd, b, n, o); atomic_fetch_add(&n_pread, 1); if (r > 0) atomic_fetch_add(&b_pread, r); return r; }
static ssize_t my_read(int fd, void* b, size_t n) { atomic_fetch_add(&n_read, 1); return read(fd, b, n); }
static ssize_t my_write(int fd, const void* b, size_t n) { ssize_t r = write(fd, b, n); atomic_fetch_add(&n_write, 1); if (r > 0) atomic_fetch_add(&b_write, r); return r; }
static ssize_t my_pwrite(int fd, const void* b, size_t n, off_t o) { atomic_fetch_add(&n_pwrite, 1); return pwrite(fd, b, n, o); }
static void* my_mmap(void* a, size_t l, int p, int f, int fd, off_t o) { atomic_fetch_add(&n_mmap, 1); return mmap(a, l, p, f, fd, o); }
static int my_munmap(void* a, size_t l) { atomic_fetch_add(&n_munmap, 1); return munmap(a, l); }
DYLD_INTERPOSE(my_fcntl, fcntl)
DYLD_INTERPOSE(my_pread, pread)
DYLD_INTERPOSE(my_read, read)
DYLD_INTERPOSE(my_write, write)
DYLD_INTERPOSE(my_pwrite, pwrite)
DYLD_INTERPOSE(my_mmap, mmap)
DYLD_INTERPOSE(my_munmap, munmap)
DYLD_INTERPOSE(my_fsync, fsync)

static long last[12];
void interpose_snapshot(const char* label) {
  long cur[12] = {atomic_load(&n_pread), atomic_load(&b_pread), atomic_load(&n_read), atomic_load(&n_write), atomic_load(&b_write), atomic_load(&n_pwrite), atomic_load(&n_mmap), atomic_load(&n_munmap), atomic_load(&fullfsync), atomic_load(&fsync_calls), 0, 0};
  fprintf(stderr, "SNAP %-28s pread=%ld (%ld B) read=%ld write=%ld (%ld B) pwrite=%ld mmap=%ld munmap=%ld F_FULLFSYNC=%ld fsync=%ld\n", label,
          cur[0]-last[0], cur[1]-last[1], cur[2]-last[2], cur[3]-last[3], cur[4]-last[4], cur[5]-last[5], cur[6]-last[6], cur[7]-last[7], cur[8]-last[8], cur[9]-last[9]);
  for (int k = 0; k < 12; k++) last[k] = cur[k];
}

__attribute__((destructor)) static void report(void) {
  fprintf(stderr, "interpose: pread=%ld (%ld bytes) read=%ld write=%ld (%ld bytes) pwrite=%ld mmap=%ld munmap=%ld\n", atomic_load(&n_pread), atomic_load(&b_pread), atomic_load(&n_read), atomic_load(&n_write), atomic_load(&b_write), atomic_load(&n_pwrite), atomic_load(&n_mmap), atomic_load(&n_munmap));
  fprintf(stderr, "interpose: F_FULLFSYNC=%ld (failed %ld) fsync=%ld other_fcntl=%ld\n",
          atomic_load(&fullfsync), atomic_load(&fullfsync_fail), atomic_load(&fsync_calls),
          atomic_load(&fcntl_other));
}

7.5 The sources of the profile driver, of the parser of `sample`'s call graph, and of the classification
($SCRATCH is the scratch directory the probe was built in).

--- profile.sh ---
#!/usr/bin/env bash
# profile.sh <probe> <label> <num> <benchmarks> <mark-or-delay> <seconds> [loops] [compact]
# Runs the probe and samples it: after a line matching the pattern <mark> appears, or after <delay> seconds
# when the mark is a number.
set -uo pipefail
S=$SCRATCH
probe=$1 label=$2 num=$3 benches=$4 mark=$5 secs=$6 loops=${7:-1} compact=${8:-}
db="$S/probe-db"; rm -rf "$db"; mkdir -p "$db"
out="$S/prof-$label"
if [ -n "$compact" ]; then export PROBE_COMPACT=1; else unset PROBE_COMPACT; fi
DYLD_INSERT_LIBRARIES="$S/interpose.dylib" "$probe" "${ENGINE:-shale}" "$num" "$db" "$benches" 0 "$loops" > "$out.out" 2>&1 &
pid=$!
if [[ "$mark" =~ ^[0-9]+$ ]]; then
  sleep "$mark"
else
  until grep -qE "$mark" "$out.out" || ! kill -0 $pid 2>/dev/null; do sleep 0.2; done
  sleep 0.3
fi
sample $pid "$secs" -file "$out.sample.txt" > /dev/null 2>&1
wait $pid
echo "$label: probe exit $?"
rm -rf "$db"
--- sample_parse.py ---
"""Summarizes the call graph of a macOS `sample` report, thread by thread.

Usage: sample_parse.py <report> [inclusive-name ...]

For each thread: its total samples, the functions at the top of the stack with 3% or more of them
(self samples: a node's count less its children's), and the inclusive samples of each named
function (summed over every node whose symbol contains the name, outermost occurrence only).
"""
import re
import sys
from collections import defaultdict

LINE = re.compile(r"^(?P<lead>[ +|!:]*)(?P<count>\d+) (?P<sym>.*)$")


def parse(path):
    threads = []
    in_graph = False
    with open(path, errors="replace") as f:
        for raw in f:
            line = raw.rstrip("\n")
            if line.startswith("Call graph:"):
                in_graph = True
                continue
            if not in_graph:
                continue
            if line.strip() == "" or line.startswith("Total number in stack"):
                if threads and line.startswith("Total number in stack"):
                    break
                continue
            m = LINE.match(line)
            if not m:
                continue
            depth = len(m.group("lead"))
            count = int(m.group("count"))
            sym = m.group("sym")
            if depth == 4:
                threads.append({"name": sym, "total": count, "nodes": []})
                continue
            if threads:
                threads[-1]["nodes"].append((depth, count, sym))
    return threads


def summarize(thread, names):
    nodes = thread["nodes"]
    selfs = defaultdict(int)
    inclusive = defaultdict(int)
    stack = []  # (depth, matched names)
    for i, (depth, count, sym) in enumerate(nodes):
        child_sum = 0
        # direct children: the next nodes at the smallest depth greater than this one
        j = i + 1
        child_depth = None
        while j < len(nodes) and nodes[j][0] > depth:
            if child_depth is None:
                child_depth = nodes[j][0]
            if nodes[j][0] == child_depth:
                child_sum += nodes[j][1]
            j += 1
        func = re.sub(r"\s+\(in [^)]*\).*$", "", sym).strip()
        selfs[func] += count - child_sum
        while stack and stack[-1][0] >= depth:
            stack.pop()
        enclosing = set()
        for _, matched in stack:
            enclosing |= matched
        matched_here = set()
        for n in names:
            if n in func:
                matched_here.add(n)
                if n not in enclosing:
                    inclusive[n] += count
        stack.append((depth, enclosing | matched_here))
    return selfs, inclusive


def main():
    path = sys.argv[1]
    names = sys.argv[2:]
    for t in parse(path):
        total = t["total"]
        if total == 0:
            continue
        selfs, inclusive = summarize(t, names)
        print(f"--- thread '{t['name'][:60]}': {total} samples")
        for func, c in sorted(selfs.items(), key=lambda kv: -kv[1]):
            if c * 100 < 3 * total:
                break
            print(f"    self {c:7d} {100.0 * c / total:5.1f}%  {func[:110]}")
        for n in names:
            c = inclusive.get(n, 0)
            print(f"    inclusive {c:7d} {100.0 * c / total:5.1f}%  {n}")


main()

"""Shares of the calling (main) thread's samples by kind of function at the top of the stack."""
import importlib.util
import re
import sys

spec = importlib.util.spec_from_file_location(
    "sp", "$SCRATCH/sample_parse.py")
sp = importlib.util.module_from_spec(spec)
sys.argv_saved = sys.argv
sys.argv = [sys.argv[0], "/dev/null"]
spec.loader.exec_module(sp)
sys.argv = sys.argv_saved

CATS = [
    ("pread syscall", r"^pread$"),
    ("write syscall (log append)", r"^write$"),
    ("F_FULLFSYNC", r"^__fcntl$"),
    ("waiting for the DB mutex", r"__psynch_mutexwait"),
    ("sleep (write slowdown)", r"__semwait_signal|nanosleep"),
    ("crc32c", r"crc32c"),
    ("malloc, free, memset", r"malloc|^free|_free|szone|nanov2|_platform_memset|bzero|operator new|operator delete|xzm"),
    ("memmove, memcmp", r"_platform_mem(move|cmp)|DYLD-STUB\$\$mem"),
    ("clock reads (harness)", r"mach_absolute_time|clock_gettime|steady_clock"),
    ("Status::ok() (out of line)", r"Status::ok"),
]


def main():
    path = sys.argv[1]
    threads = sp.parse(path)
    main_thread = next(t for t in threads if "main-thread" in t["name"])
    selfs, _ = sp.summarize(main_thread, [])
    total = main_thread["total"]
    sums = {c: 0 for c, _ in CATS}
    other = 0
    for func, c in selfs.items():
        for cat, rx in CATS:
            if re.search(rx, func):
                sums[cat] += c
                break
        else:
            other += c
    parts = [("engine and harness code", other)] + list(sums.items())
    parts = [(k, v) for k, v in parts if v > 0]
    parts.sort(key=lambda kv: -kv[1])
    print(f"{total} samples; " + "; ".join(f"{k} {100.0 * v / total:.1f}%" for k, v in parts))
    top = sorted(selfs.items(), key=lambda kv: -kv[1])[:12]
    for func, c in top:
        print(f"    self {c:6d} {100.0 * c / total:5.1f}%  {func[:110]}")


main()
